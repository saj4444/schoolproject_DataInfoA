Database setup for admin access

Replace the email below with the exact admin email used to sign in. This policy lets only that authenticated email read all saved rows; visitors still cannot read them.

grant select on table public.saved_json to authenticated;

drop policy if exists "Admin can view all saved JSON" on public.saved_json;

create policy "Admin can view all saved JSON"
on public.saved_json
for select
to authenticated
using ((select auth.jwt() ->> 'email') = 'YOUR_ADMIN_EMAIL');
